## 三、数据库设计（SQLite）

```sql
-- 空间
CREATE TABLE spaces (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  emoji TEXT NOT NULL DEFAULT '📁',
  color TEXT NOT NULL,
  sort_order INTEGER DEFAULT 0,
  created_at TEXT NOT NULL
);

-- 子分类
CREATE TABLE areas (
  id TEXT PRIMARY KEY,
  space_id TEXT NOT NULL REFERENCES spaces(id),
  name TEXT NOT NULL,
  color TEXT,
  sort_order INTEGER DEFAULT 0
);

-- 项目
CREATE TABLE projects (
  id TEXT PRIMARY KEY,
  space_id TEXT NOT NULL REFERENCES spaces(id),
  area_id TEXT REFERENCES areas(id),
  name TEXT NOT NULL,
  description TEXT,
  color TEXT NOT NULL,
  status TEXT DEFAULT 'active',
  due_date TEXT,
  created_at TEXT NOT NULL
);

-- 协作者
CREATE TABLE collaborators (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  email TEXT,
  role TEXT,
  institution TEXT,
  avatar_color TEXT NOT NULL,
  notes TEXT,
  last_sync_at TEXT,
  created_at TEXT NOT NULL
);

-- 标签
CREATE TABLE tags (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  color TEXT NOT NULL
);

-- 任务
CREATE TABLE tasks (
  id TEXT PRIMARY KEY,
  title TEXT NOT NULL,
  description TEXT,
  space_id TEXT NOT NULL REFERENCES spaces(id),
  area_id TEXT REFERENCES areas(id),
  project_id TEXT REFERENCES projects(id),
  paper_id TEXT REFERENCES papers(id),
  priority TEXT NOT NULL DEFAULT 'medium',
  status TEXT NOT NULL DEFAULT 'inbox',
  start_date TEXT,
  due_date TEXT,
  estimated_hours REAL,
  actual_hours REAL DEFAULT 0,
  subtasks TEXT DEFAULT '[]',        -- JSON array
  tag_ids TEXT DEFAULT '[]',         -- JSON array
  collaborator_ids TEXT DEFAULT '[]',-- JSON array
  dependency_ids TEXT DEFAULT '[]',  -- JSON array
  recurrence TEXT,                   -- JSON: {type, interval, days}
  reminder_days TEXT DEFAULT '[]',   -- JSON array [1, 3, 7]
  is_inbox INTEGER DEFAULT 1,
  raw_input TEXT,
  ai_parsed INTEGER DEFAULT 0,
  completed_at TEXT,
  created_at TEXT NOT NULL,
  updated_at TEXT NOT NULL
);

-- 时间追踪记录
CREATE TABLE time_logs (
  id TEXT PRIMARY KEY,
  task_id TEXT NOT NULL REFERENCES tasks(id),
  started_at TEXT NOT NULL,
  ended_at TEXT,
  duration_minutes INTEGER,
  is_pomodoro INTEGER DEFAULT 0
);

-- 论文
CREATE TABLE papers (
  id TEXT PRIMARY KEY,
  project_id TEXT REFERENCES projects(id),
  title TEXT NOT NULL,
  overleaf_url TEXT,
  overleaf_project_id TEXT,
  target_journal_ids TEXT DEFAULT '[]',
  submission_status TEXT DEFAULT 'preparing',
  submission_deadline TEXT,
  collaborator_ids TEXT DEFAULT '[]',
  current_word_count INTEGER DEFAULT 0,
  notes TEXT,
  created_at TEXT NOT NULL,
  updated_at TEXT NOT NULL
);

-- 论文版本快照
CREATE TABLE paper_snapshots (
  id TEXT PRIMARY KEY,
  paper_id TEXT NOT NULL REFERENCES papers(id),
  version_label TEXT,
  snapshot_at TEXT NOT NULL,
  word_count INTEGER,
  diff_summary TEXT,
  file_path TEXT,
  notes TEXT
);

-- 期刊库
CREATE TABLE journals (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  abbreviation TEXT NOT NULL,
  field TEXT,
  impact_factor REAL,
  deadline TEXT,
  website TEXT,
  is_preset INTEGER DEFAULT 1
);

-- 会议
CREATE TABLE meetings (
  id TEXT PRIMARY KEY,
  title TEXT NOT NULL,
  space_id TEXT REFERENCES spaces(id),
  project_id TEXT REFERENCES projects(id),
  scheduled_at TEXT NOT NULL,
  duration_minutes INTEGER DEFAULT 60,
  attendee_ids TEXT DEFAULT '[]',
  agenda TEXT,
  notes TEXT,
  action_items TEXT DEFAULT '[]',
  recurrence TEXT,
  created_at TEXT NOT NULL
);

-- 通知
CREATE TABLE notifications (
  id TEXT PRIMARY KEY,
  type TEXT NOT NULL,
  title TEXT NOT NULL,
  body TEXT,
  related_id TEXT,
  related_type TEXT,
  is_read INTEGER DEFAULT 0,
  created_at TEXT NOT NULL
);

-- 用户配置
CREATE TABLE config (
  key TEXT PRIMARY KEY,
  value TEXT NOT NULL
);
```

## 五、AI 服务设计（智谱 GLM-4）

### 5.1 语义解析 Prompt

```python
PARSE_SYSTEM_PROMPT = """
你是一个任务解析助手。将用户输入解析为结构化任务 JSON。

输出格式（严格 JSON）：
{
  "title": "任务标题",
  "space": "academic|work|life",
  "area": "子分类（可选）",
  "priority": "urgent|high|medium|low",
  "due_date": "YYYY-MM-DD（可选）",
  "start_date": "YYYY-MM-DD（可选）",
  "collaborators": ["姓名列表"],
  "subtasks": ["子任务1", "子任务2"],
  "tags": ["标签"],
  "estimated_hours": 数字（可选）,
  "is_meeting": true/false,
  "meeting_time": "YYYY-MM-DD HH:MM（会议时专用）"
}

规则：
- 今天是 {today}
- 判断优先级时考虑截止时间紧迫性和任务描述中的关键词
- 提取所有提到的人名作为协作者
- 自动生成合理的子任务（2-4个）
"""
```

### 5.2 定时任务（APScheduler）

```python
# 每天 07:50 - 生成今日计划通知
scheduler.add_job(generate_daily_plan, 'cron', hour=7, minute=50)

# 每天 08:00 - 发送每日摘要邮件
scheduler.add_job(send_daily_digest, 'cron', hour=8, minute=0)

# 每天 02:00 - Overleaf 同步所有论文
scheduler.add_job(sync_all_papers, 'cron', hour=2, minute=0)

# 每天 09:00 - 检查 deadline 并发预警邮件
scheduler.add_job(check_deadlines, 'cron', hour=9, minute=0)

# 每周日 20:00 - 发送周回顾邮件
scheduler.add_job(send_weekly_review, 'cron', day_of_week='sun', hour=20)

# 每月 1 日 09:00 - 发送月度复盘邮件
scheduler.add_job(send_monthly_report, 'cron', day=1, hour=9)

# 每小时 - 检查协作者同步状态
scheduler.add_job(check_collaborator_sync, 'interval', hours=1)

# 每 5 分钟 - 检查即将到期任务（实时播报）
scheduler.add_job(check_urgent_tasks, 'interval', minutes=5)
```

## 七、启动方式

### 一键启动脚本（start.sh）

```bash
#!/bin/bash
# 启动后端
cd backend
source .venv/bin/activate
uvicorn main:app --host 0.0.0.0 --port 8000 &
BACKEND_PID=$!

# 启动前端
cd ../frontend
npm run dev &
FRONTEND_PID=$!

echo "✅ Life OS 已启动"
echo "   前端: http://localhost:5173"
echo "   后端: http://localhost:8000"
echo "   API文档: http://localhost:8000/docs"
echo ""
echo "按 Ctrl+C 停止所有服务"

wait
```

### 安装依赖

```bash
# 后端
cd backend
python3.12 -m venv .venv
.venv/bin/pip install -r requirements.txt
cp .env.example .env
# 编辑 .env 填入配置

# 前端
cd frontend
npm install
```
